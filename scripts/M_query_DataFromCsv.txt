// this query is used in TestStatistics.xlsx report

let
    // get the path to statistics .csv file
    Params = Excel.CurrentWorkbook(){[Name=tblSettings]}[Content],
    #Changed Type = Table.TransformColumnTypes(Params,{{Parameter, type text}, {Value, type text}, {Description, type text}}),
    statsFilePath = #Changed Type{0}[Value],

    // Solutio Codes
    SolutionCode = Excel.CurrentWorkbook(){[Name=tblSolutionCode]}[Content],
    #Change SolutionCode Type = Table.TransformColumnTypes(SolutionCode,{{Code, Int64.Type}, {String Value, type text}}),
    #Renamed Columns = Table.RenameColumns(#Change SolutionCode Type,{{String Value, Return Value}}),

    // read the file
    Source = Csv.Document(File.Contents(statsFilePath),[Delimiter=;, Columns=81, Encoding=1252, QuoteStyle=QuoteStyle.None]),
    #Promoted Headers = try Table.PromoteHeaders(Source, [PromoteAllScalars=true]),
    FileNotFound = #Promoted Headers[HasError],  returns a record [HasError truefalse, Value table]

    // transform columns
    DecimalCols = { Overall Wall Clock Time (s),  Solution value,  Solver Wall Clock Time (s),  PNB_TREE_WALK_STRATEGY_EPSILON,  PNB_CUT_CONSOLIDATION_PARTIAL_THRESHOLD,  PNB_LEVEL_LAMBDA,  PNB_ODA_KAPPA,  PNB_TOLERANCE_ZERO,  PNB_TOLERANCE_INTEGER,  PNB_EPSILON_ABSOLUTE,  PNB_EPSILON_RELATIVE,  PNB_CVAR_BETA,  PNB_MAXIMAL_SOLUTION_TIME,  Time Setup,  Time Stage 0,  Time Stage 1,  Time Stage 2,  Time Stage 3,  Time Stage 4,  Time Stage 5,  Time Stage 6,  Time Stage 7,  Time Stage 8,  Time Stage 9,  Time Stage 10,  Time Master,  Time Projection,  Time ODA,  Time Sub,  Build Master,  Build Projection,  Build ODA,  Build Sub,  Generate Cut},
    IntegerCols = { Iterations,  Return Code,  Infeasibility Iterations,  Substantial Iterations,  Insubstantial Iterations,  Critical Iterations,  Max Length between Crit. It., Created Optimality Cuts,  Created Feasibility Cuts,  PNB_BENDERS,  PNB_WARM_START,  PNB_LP_METHOD,  PNB_FEAS_CUT_LP_METHOD,  PNB_AGGREGATED_OPTIMALITY_CUTS,  PNB_RELAXATION,  PNB_TREE_WALK_STRATEGY,  PNB_TREE_WALK_STRATEGY_STAGE,  PNB_HANDLE_FEAS_CUTS,  PNB_STAGE_AGGREGATION_START,  PNB_STAGE_AGGREGATION_END,  PNB_CUT_CONSOLIDATION_STRIKE_THRESHOLD,  PNB_USE_START_SOLUTION,  PNB_NUM_CPU,  PNB_CPU_USED,  PNB_ODA_AGGREGATES,  PNB_STAGE_LEVEL,  PNB_CVAR_RHO,  PNB_COMPUTE_PRIMAL_SOLUTIONS,  PNB_COMPUTE_DUAL_SOLUTIONS,  PNB_LP_ROUNDS_LIMIT},
    BooleanCols = { PNB_PRINT_SUBPROBLEMS,  PNB_DYNAMIC_AGGREGATION,  PNB_STAGE_AGGREGATION,  PNG_STAGE_AGGREGATION_DIRECTION,  PNB_BOUNDS_IN_MATRIX,  PNB_CUT_CONSOLIDATION,  PNB_CUT_CONSOLIDATION_CUT_COMPUTATION,  PNB_CVAR_WITH_FIRST_STAGE,  PNB_COMPUTE_OBJECTIVE_VALUES},
    #. to comma = Table.ReplaceValue(#Promoted Headers[Value],.,,,Replacer.ReplaceText,DecimalCols),
    DecimalColsType = Table.TransformColumnTypes(#. to comma, List.Transform(DecimalCols, each {_, type number})),
    IntegerColsType = Table.TransformColumnTypes(DecimalColsType, List.Transform(IntegerCols, each {_, Int64.Type})),
    BooleanColsType = Table.TransformColumnTypes(IntegerColsType, List.Transform(BooleanCols, each {_, type logical})),
    TimeColsType = Table.TransformColumnTypes(BooleanColsType,{{ Time, type time}}),
    DateColsType = Table.TransformColumnTypes(TimeColsType, {{Date, type date}}, en-US),
    ReturnValue = Table.Join(DateColsType,  Return Code, #Renamed Columns, Code),
    #Reordered Columns = Table.ReorderColumns(ReturnValue,{Problem Name,  Overall Wall Clock Time (s),  Iterations,  Solution value,  Return Code, Return Value,  Infeasibility Iterations,  Substantial Iterations,  Insubstantial Iterations,  Solver Wall Clock Time (s),  Critical Iterations,  Max Length between Crit. It., Created Optimality Cuts,  Created Feasibility Cuts,  PNB_PRINT_SUBPROBLEMS,  PNB_BENDERS,  PNB_WARM_START,  PNB_LP_METHOD,  PNB_FEAS_CUT_LP_METHOD,  PNB_AGGREGATED_OPTIMALITY_CUTS,  PNB_DYNAMIC_AGGREGATION,  PNB_RELAXATION,  PNB_TREE_WALK_STRATEGY,  PNB_TREE_WALK_STRATEGY_STAGE,  PNB_TREE_WALK_STRATEGY_EPSILON,  PNB_HANDLE_FEAS_CUTS,  PNB_STAGE_AGGREGATION,  PNB_STAGE_AGGREGATION_START,  PNB_STAGE_AGGREGATION_END,  PNG_STAGE_AGGREGATION_DIRECTION,  PNB_BOUNDS_IN_MATRIX,  PNB_CUT_CONSOLIDATION,  PNB_CUT_CONSOLIDATION_CUT_COMPUTATION,  PNB_CUT_CONSOLIDATION_STRIKE_THRESHOLD,  PNB_CUT_CONSOLIDATION_PARTIAL_THRESHOLD,  PNB_USE_START_SOLUTION,  PNB_NUM_CPU,  PNB_CPU_USED,  PNB_MAXIMAL_SOLUTION_TIME,  PNB_LEVEL_LAMBDA,  PNB_ODA_KAPPA,  PNB_ODA_AGGREGATES,  PNB_STAGE_LEVEL,  PNB_TOLERANCE_ZERO,  PNB_TOLERANCE_INTEGER,  PNB_EPSILON_ABSOLUTE,  PNB_EPSILON_RELATIVE,  PNB_CVAR_BETA,  PNB_CVAR_RHO,  PNB_CVAR_WITH_FIRST_STAGE,  PNB_COMPUTE_PRIMAL_SOLUTIONS,  PNB_COMPUTE_DUAL_SOLUTIONS,  PNB_COMPUTE_OBJECTIVE_VALUES,  PNB_LP_ROUNDS_LIMIT,  Time Setup,  Time Stage 0,  Time Stage 1,  Time Stage 2,  Time Stage 3,  Time Stage 4,  Time Stage 5,  Time Stage 6,  Time Stage 7,  Time Stage 8,  Time Stage 9,  Time Stage 10,  Time Master,  Time Projection,  Time ODA,  Time Sub,  Build Master,  Build Projection,  Build ODA,  Build Sub,  Generate Cut, Date,  Time,  Min_CPU,  Min_Solver,  Max_CPU,  Max_Solver, , Code}),
    #Removed Columns = Table.RemoveColumns(#Reordered Columns,{Code}),
    Result = if FileNotFound then error The .csv file with statistics not found. Check the statFilePath on Settings tab. else #Removed Columns
in
    Result
